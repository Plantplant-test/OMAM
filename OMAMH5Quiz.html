            scores = { gold: 0, wood: 0, water: 0, fire: 0, earth: 0 };

           

            screenWelcome.classList.add('hidden');

            screenQuiz.classList.remove('hidden');

           

            showQuestion();

        }

        // --- 核心函數：顯示問題 ---

        function showQuestion() {

            const q = questions[currentQuestionIndex];

           

            // 更新進度條

            const progressPercent = ((currentQuestionIndex + 1) / questions.length) * 100;

            quizProgressText.innerText = `問題 ${currentQuestionIndex + 1} / ${questions.length}`;

            quizProgressPercent.innerText = `${Math.round(progressPercent)}%`;

            quizProgressBar.style.width = `${progressPercent}%`;

            // 渲染題目

            quizQuestion.innerText = q.question;

            // 渲染選項

            quizOptions.innerHTML = '';

            q.options.forEach((opt, idx) => {

                const btn = document.createElement('button');

                btn.className = "w-full text-left bg-slate-50 hover:bg-emerald-50/50 border border-slate-200/80 hover:border-emerald-300 rounded-2xl p-4 text-sm text-slate-700 transition-all duration-200 transform active:scale-[0.99] flex items-start gap-3 shadow-sm";

               

                // 選項序號

                const numSpan = document.createElement('span');

                numSpan.className = "flex items-center justify-center w-5 h-5 rounded-full bg-slate-200 text-slate-600 text-xs font-bold shrink-0 mt-0.5";

                numSpan.innerText = String.fromCharCode(65 + idx); // A, B, C...

                const textSpan = document.createElement('span');

                textSpan.className = "leading-relaxed";

                textSpan.innerText = opt.text;

                btn.appendChild(numSpan);

                btn.appendChild(textSpan);

                // 綁定點擊事件

                btn.addEventListener('click', () => selectOption(opt.element));

                quizOptions.appendChild(btn);

            });

        }

        // --- 核心函數：點選選項 ---

        function selectOption(element) {

            scores[element]++;

           

            if (currentQuestionIndex < questions.length - 1) {

                currentQuestionIndex++;

                // 增加淡入過渡感覺

                quizOptions.style.opacity = 0;

                setTimeout(() => {

                    showQuestion();

                    quizOptions.style.opacity = 1;

                }, 150);

            } else {

                calculateAndShowResult();

            }

        }

        // --- 核心函數：計算並展示結果 ---

        function calculateAndShowResult() {

            screenQuiz.classList.add('hidden');

            screenLoading.classList.remove('hidden');

            // 模擬分析過程的動態文案

            const phrases = [

                { title: "正在調配五行能量...", sub: "分析您的職場生存本能與基因強度" },

                { title: "正在掃描辦公桌氣場...", sub: "評估您周邊的內耗指數與壓力值" },

                { title: "正在挑選命定守護植...", sub: "正在 Plant Plant 溫室中為您進行基因配對" }

            ];

            let step = 0;

            const interval = setInterval(() => {

                step++;

                if (step < phrases.length) {

                    loadingTitle.innerText = phrases[step].title;

                    loadingSub.innerText = phrases[step].sub;

                }

            }, 800);

            // 計算得分最高之五行屬性

            setTimeout(() => {

                clearInterval(interval);

               

                let maxScore = -1;

                let finalElement = 'gold';

                for (const key in scores) {

                    if (scores[key] > maxScore) {

                        maxScore = scores[key];

                        finalElement = key;

                    }

                }

               

                selectedElement = finalElement;

                renderResult(finalElement);

            }, 2400);

        }

        // --- 核心函數：渲染結果畫面 ---

            // 1. 設置基本資訊

            resTitle.innerText = data.title;

            resSubtitle.innerText = data.subtitle;

            resDesc.innerText = data.desc;

            resAdvantage.innerText = data.advantage;

            resDisadvantage.innerText = data.disadvantage;

           

            resPlantName.innerText = data.plantName;

            resPlantDesc.innerText = data.plantDesc;

            resPlantPrice.innerText = data.price;

            couponCode.innerText = data.coupon;

            // 2. 注入專屬植物 SVG 插畫

            resPlantSvgContainer.innerHTML = data.svg;

            // 3. 根據五行屬性動態調整卡片與按鈕色系

            resThemeBanner.className = `rounded-2xl p-5 text-white text-left relative overflow-hidden mb-4 shadow-md bg-gradient-to-r ${data.themeClass}`;

            btnBuy.className = `bg-gradient-to-r ${data.themeClass} hover:opacity-90 text-white font-bold py-3.5 px-4 rounded-xl text-sm transition transform active:scale-[0.98] shadow-md flex items-center justify-center gap-1`;

            // 隱藏加載頁，展示結果頁

            screenLoading.classList.add('hidden');

            screenResult.classList.remove('hidden');

        }

        // --- 輔助函數：顯示自定義 Toast 提示（不使用內置 alert） ---

            }, 2000);

        }
